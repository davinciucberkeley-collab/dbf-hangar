### A5. Allow-list script (amended): replace the script in Pass check 1 with this one
Fenced blocks are allowed only after the line `## Appendix`. Fences tagged `text` are skipped. A fence tagged `json` must parse as JSON and contain `plane-mcp.exe` (the Desktop config). A fence before the Appendix that is not tagged `text` is an error (this is how "no commands in the body" is enforced). Save as `.logs/check_guide.py`:
```python
import json, re, sys
from pathlib import Path
root = Path("C:/Users/trist/wt/u4.17-team-guide")
brief = (root / "plans/2026-11-20-standardization/briefs/U4.17-team-guide.md").read_text(encoding="utf-8")
guide = (root / "docs/team-guide.md").read_text(encoding="utf-8")

def norm(line):
    line = line.strip()
    line = re.sub(r"^(\$|PS>)\s+", "", line)
    line = re.sub(r"\s+#\s.*$", "", line)
    line = re.sub(r"<[^>]*>", "<>", line)
    return re.sub(r"\s+", " ", line)

def fences(text):
    out, cur, tag, start = [], None, None, 0
    for i, ln in enumerate(text.splitlines()):
        if ln.startswith("```"):
            if cur is None:
                cur, tag, start = [], ln[3:].strip(), i
            else:
                out.append((tag, cur, start)); cur = None
        elif cur is not None:
            cur.append(ln)
    return out

block = brief.split("<!-- verified" + ":start -->")[1].split("<!-- verified" + ":end -->")[0]
allowed = {norm(l) for _, ls, _ in fences(block) for l in ls if l.strip()}
allowed.add(norm('plane design propose designs/<id> "<title>"'))
lines = guide.splitlines()
idx = [i for i, l in enumerate(lines) if l.startswith("## Appendix")]
app = idx[0] if idx else 10**9
bad, n = [], 0
for tag, ls, start in fences(guide):
    if tag == "text":
        continue
    if tag == "json":
        try:
            json.loads("\n".join(ls))
        except Exception as e:
            bad.append(f"json fence at line {start+1} does not parse: {e}")
        if "plane-mcp.exe" not in "\n".join(ls):
            bad.append(f"json fence at line {start+1} lacks plane-mcp.exe")
        continue
    if start < app:
        bad.append(f"command fence before the Appendix at line {start+1}")
        continue
    for l in ls:
        if not l.strip() or l.strip().startswith("#"):
            continue
        n += 1
        if norm(l) not in allowed:
            bad.append("NOT ALLOWED: " + l)
for b in bad:
    print(b)
print(f"{n} command lines checked, {len(bad)} problems, guide has {len(lines)} lines")
if len(lines) > 700:
    print("TOO LONG"); sys.exit(1)
if bad or not idx:
    print("no Appendix heading" if not idx else ""); sys.exit(1)
print("OK")
```
