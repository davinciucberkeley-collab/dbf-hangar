## Data: `ledger/items.yaml`
Top level `{schema: "ledger/1", items: [...]}`. Each item:
`{id: str, kind: ruling|task|idea|decision, text: str, source: str, owner: str, status: open|done|dropped|superseded|decided, due: "" or YYYY-MM-DD, units: [str], evidence: str, decided_in: str, updated: YYYY-MM-DD}`.
`schema/ledger.schema.json` is the JSON Schema for this file (draft 2020-12, `additionalProperties: false` on items, `id` pattern `^(R|T|I|D)\d+$`).

## `core/hangar/ledger.py`
```python
class Ledger:
    def __init__(self, path: Path): ...                   # keeps it as self.path; missing file = empty ledger
    items: list[Item]                                      # Item: a dataclass with the fields above
    def add(self, kind, text, source, owner="", item_id=None, due="") -> str   # next id per prefix R/T/I/D unless item_id given; saves
    def get(self, item_id) -> Item                         # KeyError if absent
    def set(self, item_id, status, evidence="", decided_in="") -> None  # rules below; saves
    def check(self, repo: Path) -> list[Issue]             # Issue(item_id: str, kind: str, message: str)
    def report_overdue(self, today: str) -> list[str]      # ids with status open and due < today
    def report(self) -> str                                # markdown, see CLI
    def seed_from_plan(self, plan: Path) -> int            # see below; returns items added or updated
```
Rules for `set`: `done` needs non-empty `evidence`; `dropped` and `superseded` need `decided_in`; `decided` needs `evidence` (the decision taken); unknown status → `ValueError`; unknown id → `KeyError`. Saving uses `ruamel.yaml` round-trip via `hangar.yamlio` and validates against the schema before writing.
`check`: an item `done` whose `evidence` is neither a URL (`http://`/`https://`) nor an existing repo-relative path nor a 7–40 hex commit id → `Issue(kind="missing-evidence")`; `dropped` without `decided_in` → `Issue(kind="dropped-without-decision")`.
`seed_from_plan`: reads the plan's §1 table (rows `| R<n> | who | requirement | units |`) into `ruling` items (`id` = R<n>, `text` = requirement, `source` = "plan §1; " + who, `units` = every `U<d>.<d>` in the last cell) and the §3 table (`| D<n> | question | default | ask | by | status |`) into `decision` items (`text` = question, `due` = "2026-" + by, `status` = `decided` with `evidence` = "default: " + default when the status cell starts with "decided", else `open`). Existing ids are updated in place (text, units, due), never duplicated; statuses already set by hand are kept.

## CLI (`plane ledger …`; exit 0 ok, 1 check failed, 2 bad input)
- `seed --plan PATH` → prints `seeded N`
- `add KIND TEXT --source S [--owner O] [--due D]` → prints the new id
- `set ID STATUS [--evidence E] [--decided-in X]`
- `check` → one line per issue, exit 1 if any
- `report [--today YYYY-MM-DD]` → markdown: counts per kind and status, the overdue list, and `Open prompts: N` read from `ledger/prompts.jsonl` (count of ids whose latest line has `status == "open"`; missing file = 0)

## Exact tests (`tests/test_ledger.py`; write first, see them fail)
```python
from pathlib import Path
import pytest
from click.testing import CliRunner
from hangar.cli import cli
from hangar.ledger import Ledger

REPO = Path(__file__).resolve().parents[1]
PLAN = REPO / "plans" / "2026-11-20-standardization.md"

@pytest.fixture
def led(tmp_path): return Ledger(tmp_path / "items.yaml")

def test_ids_per_prefix(led):
    assert led.add("task", "a", source="s") == "T1"
    assert led.add("task", "b", source="s") == "T2"
    assert led.add("idea", "c", source="s") == "I1"

def test_dropped_needs_decision_and_is_not_overdue(led):
    t = led.add("task", "OpenCAE GUI port", source="2026-11-11#61:02", due="2026-11-15")
    with pytest.raises(ValueError, match="decided_in"):
        led.set(t, "dropped")
    led.set(t, "dropped", decided_in="2026-11-18#26:16")
    assert t not in led.report_overdue("2026-11-20")

def test_open_past_due_is_overdue(led):
    t = led.add("task", "x", source="s", due="2026-11-19")
    assert led.report_overdue("2026-11-20") == [t]

def test_done_needs_evidence_that_exists(led):
    t = led.add("task", "x", source="s")
    led.set(t, "done", evidence="designs/does-not-exist/plane.yaml")
    assert [i.kind for i in led.check(REPO)] == ["missing-evidence"]
    led.set(t, "done", evidence="https://github.com/davinciucberkeley-collab/dbf-hangar/pull/1")
    assert led.check(REPO) == []

def test_seed_from_plan_has_R1_to_R40_and_D1_to_D10(led):
    led.seed_from_plan(PLAN)
    ids = {i.id for i in led.items}
    assert {f"R{n}" for n in range(1, 41)} <= ids and {f"D{n}" for n in range(1, 11)} <= ids
    assert led.get("D10").status == "open" and led.get("D1").status == "decided"
    assert "U1.3" in led.get("R1").units

def test_seed_is_idempotent_and_keeps_manual_status(led):
    led.seed_from_plan(PLAN); led.set("R1", "done", evidence="https://example.com/pr/1")
    led.seed_from_plan(PLAN)
    assert len([i for i in led.items if i.id == "R1"]) == 1 and led.get("R1").status == "done"

def test_file_validates_against_schema(led):
    import json, jsonschema, yaml
    led.seed_from_plan(PLAN)
    data = yaml.safe_load(led.path.read_text(encoding="utf-8"))
    jsonschema.validate(data, json.loads((REPO / "schema/ledger.schema.json").read_text(encoding="utf-8")))

def test_cli_report_counts_open_prompts(tmp_path, monkeypatch):
    monkeypatch.chdir(tmp_path)
    (tmp_path / "ledger").mkdir()
    (tmp_path / "ledger/prompts.jsonl").write_text('{"id":"P1","status":"open","text":"a","session":"s"}\n', encoding="utf-8")
    r = CliRunner().invoke(cli, ["ledger", "report", "--today", "2026-11-20"])
    assert r.exit_code == 0 and "Open prompts: 1" in r.output

def test_committed_ledger_is_seeded():
    led = Ledger(REPO / "ledger" / "items.yaml")
    assert led.get("R39").kind == "ruling"
```
