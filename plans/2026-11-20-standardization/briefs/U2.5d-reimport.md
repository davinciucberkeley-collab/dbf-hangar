```python
"""Replay a design's CHANGELOG.yaml onto a freshly imported plane.yaml with the design.set op (the op `plane design set` calls).
usage: python replay.py <design-folder> <old-CHANGELOG.yaml>
Entries with one date stamp are one atomic call (`also`), as when first made. Per group:
  remove: skipped if the element is already absent;  add/modify: skipped if the value is already equal;
  an add/modify whose key holds a DIFFERENT value, or any call the schema rejects, is a CONFLICT: printed, exit 1."""
import itertools, json, sys
import yaml
from hangar.ops import call
folder, changelog = sys.argv[1], sys.argv[2]
entries = yaml.safe_load(open(changelog, encoding="utf-8"))
entries.sort(key=lambda x: x["date"])          # stable: file order inside one stamp
MISSING = object()
def current(key):
    try:
        r = call("design.get", path=folder, key=key)
        return r.data["value"] if r.ok else MISSING
    except Exception:
        return MISSING
def val(e):
    v = e["new"]
    return json.dumps(v) if isinstance(v, (dict, list)) else str(v)
# entries the importer now subsumes: log key -> (new key, new value). The importer emits every harness itself
# (E-HRN-13 as id hrn-bat, no gauge), so the logged `wiring` add (D21: 14 AWG) is replayed as the gauge alone.
REWRITE = {"wiring": ("wiring.harnesses[hrn-bat].gauge_awg", 14)}
for e in entries:
    if e["key"] in REWRITE and current(e["key"]) is not MISSING:
        e["key"], e["new"] = REWRITE[e["key"]]
conflicts = 0
for date, grp in itertools.groupby(entries, key=lambda x: x["date"]):
    grp = list(grp)
    kind = {g["kind"] for g in grp}.pop()
    todo = []
    for g in grp:
        cur = current(g["key"])
        if kind == "remove":
            if cur is not MISSING:
                todo.append(g)
        elif cur is MISSING:
            todo.append(g)
        elif cur != g["new"]:
            print("CONFLICT", date, g["key"], "holds", cur, "log says", g["new"]); conflicts += 1
        # equal: skip
    if not todo:
        print("SKIP    ", date, kind, [g["key"] for g in grp]); continue
    why = "replay of " + date + " (U2.5d re-import): " + todo[0]["why"]
    kw = dict(path=folder, key=todo[0]["key"], why=why, kind=kind)
    if kind != "remove":
        kw["value"] = val(todo[0])
    kw["also"] = [g["key"] + "=" + ("" if kind == "remove" else val(g)) for g in todo[1:]] or None
    try:
        r = call("design.set", **kw)
    except Exception as ex:                      # the op raises OpInputError on a bad key or schema failure
        print("CONFLICT", date, [g["key"] for g in todo], ex); conflicts += 1; continue
    print("APPLIED " if r.ok else "CONFLICT", date, kind, [g["key"] for g in todo])
    conflicts += 0 if r.ok else 1
sys.exit(1 if conflicts else 0)
```
