## The expressions (what the script below writes)

- Part mesh components (50 or 26 minus 2 tubes): `components[<id>].position[1]` = the offset (all are `position: [0, 0, 0]` today; the script asserts it).
- Fin tubes: `components[fin-001-l].geometry.params.center[1]` = `-($span_mm/2 - 70)` (stored -380), `components[fin-001-r].geometry.params.center[1]` = `$span_mm/2 - 70` (stored 380).
- Every joint (attachment) whose parent or child is a moved part (the only unmoved parents are wng-001 and wng-002): `attachments[<id>].location[1]` = offset FIRST, then the stored number (`450 - $span_mm/2 - 380.416`, `$span_mm/2 - 450 + 128.599`), so span 900 reproduces the stored value exactly; for the 4 articulated joints E-WNG-10, E-WNG-11, E-MEC-02, E-MEC-07 also `attachments[<id>].articulation.pivot[1]` (variant A; in B only those whose ends are moved). The side of a joint is the side of its child if the child moved, else of its parent.
- Pitot (avn-005, avn-006, avn-009-s, avn-009-t) is on the left (y < 0) and moves with the left offset.

```python
"""Emit (or run with --run) the design.set calls that convert concept-a-param to span expressions (T129).
usage: python u4.29-data-gen.py <design_dir> <A|B> [--run]"""
import subprocess, sys
from pathlib import Path
import trimesh, yaml

design, variant = Path(sys.argv[1]), sys.argv[2]
run = "--run" in sys.argv
PY = "C:/Users/trist/dbf-hangar/.venv/Scripts/python.exe"
L, R = "450 - $span_mm/2", "$span_mm/2 - 450"
pitot = ["avn-005", "avn-006", "avn-009-s", "avn-009-t"]
fins = ([f"fin-00{i}-{s}" for i in range(2, 7) for s in "lr"] + [f"ger-00{i}-{s}" for i in (1, 2) for s in ("ll", "lr", "ul", "ur")]
        + ["wng-005-l", "wng-005-r"])
elev = ([f"{p}-{s}" for p in ("wng-006", "wng-007", "wng-009", "wng-010", "mec-001", "mec-002", "mec-003", "mec-004", "mec-005", "hwc-002") for s in "lr"]
        + ["wng-008-ll", "wng-008-lu", "wng-008-rl", "wng-008-ru"])
tubes = ["fin-001-l", "fin-001-r"]
moved = pitot + fins + tubes + (elev if variant == "A" else [])
d = yaml.safe_load((design / "plane.yaml").read_text(encoding="utf-8"))
comp = {c["id"]: c for c in d["components"]}
side = {}
for cid in moved:
    c = comp[cid]
    if cid in tubes:
        side[cid] = "l" if c["geometry"]["params"]["center"][1] < 0 else "r"
    else:
        m = trimesh.load(design / c["geometry"]["mesh"]["file"])
        side[cid] = "l" if m.bounds.mean(axis=0)[1] < 0 else "r"
        assert c["position"] == [0, 0, 0], cid
off = lambda s: L if s == "l" else R
def plus(s, v):
    return f"{off(s)} + {v}" if v >= 0 else f"{off(s)} - {abs(v)}"
pairs = []
for cid in moved:
    if cid in tubes:
        assert abs(comp[cid]["geometry"]["params"]["center"][1]) == 380
        pairs.append((f"components[{cid}].geometry.params.center[1]",
                      "-($span_mm/2 - 70)" if side[cid] == "l" else "$span_mm/2 - 70"))
    else:
        pairs.append((f"components[{cid}].position[1]", off(side[cid])))
njoints = 0
for a in d["attachments"]:
    s = side.get(a["child"]) or side.get(a["parent"])
    if not s:
        continue
    njoints += 1
    pairs.append((f"attachments[{a['id']}].location[1]", plus(s, a["location"][1])))
    if a.get("articulation"):
        pairs.append((f"attachments[{a['id']}].articulation.pivot[1]", plus(s, a["articulation"]["pivot"][1])))
print(f"variant {variant}: {len(moved)} components, {njoints} joints, {len(pairs)} pairs", file=sys.stderr)
for i in range(0, len(pairs), 20):
    chunk = pairs[i:i + 20]
    cmd = [PY, "-c", "import sys; from hangar.cli import main; sys.argv=['plane']+sys.argv[1:]; main()", "design", "set",
           str(design / "plane.yaml"), chunk[0][0], "--value", chunk[0][1], "--why",
           "T129: placement follows parameters.span_mm"] + [x for k, v in chunk[1:] for x in ("--also", f"{k}={v}")]
    if run:
        r = subprocess.run(cmd, capture_output=True, text=True)
        print(i, r.returncode, r.stdout.strip()[-300:], r.stderr.strip()[-300:])
        if r.returncode:
            sys.exit(1)
    else:
        print(chunk[0], "...", len(chunk))
```

```python
"""Test 5: span demo on a scratch copy. usage: python u4.29-data-check.py <A|B> ; prints OK or raises."""
import shutil, sys, tempfile
from pathlib import Path
import yaml
from hangar.checks import tier1_geometry
from hangar.design import load_design, load_standards
from hangar.geometry import component_parts
from hangar.ops import call

SRC = Path("C:/Users/trist/wt/u4.29-data/designs/concept-a-param")
variant = sys.argv[1]
span = "820" if variant == "A" else "980"
dst = Path(tempfile.mkdtemp()) / "p"
shutil.copytree(SRC, dst)
raw = yaml.safe_load((dst / "plane.yaml").read_text(encoding="utf-8"))
conv = [c["id"] for c in raw["components"] if isinstance(c.get("position"), list) and isinstance(c["position"][1], str)]
print(len(conv), "converted mesh components")
assert len(conv) == (48 if variant == "A" else 24), len(conv)

def ys(design):
    out = {}
    for cid in conv:
        c = next(c for c in design.data["components"] if c["id"] == cid)
        bs = [p.bounds for p in component_parts(c, design.base_dir)]
        out[cid] = (min(b[0][1] for b in bs), max(b[1][1] for b in bs))
    return out

def rows(design):
    rep = tier1_geometry(design, load_standards())
    return sorted(f"{r.status} {r.check_id} {r.message}" for r in rep.results if r.status in ("FAIL", "WARN"))

d0 = load_design(dst)
y0, r0 = ys(d0), rows(d0)
print("span 900 FAIL/WARN rows:", len(r0))
r = call("design.set", path=str(dst), key="parameters.span_mm", value=span, why="T129 demo")
assert r.ok, r.error if hasattr(r, "error") else r
print("\n".join(r.messages[-6:]))
d1 = load_design(dst)
y1 = ys(d1)
delta = (900 - int(span)) / 2                     # 40 for 820, -40 for 980 (left parts move +delta in y)
for cid in conv:
    left = (y0[cid][0] + y0[cid][1]) / 2 < 0
    want = delta if left else -delta
    assert abs((y1[cid][0] - y0[cid][0]) - want) < 1e-6 and abs((y1[cid][1] - y0[cid][1]) - want) < 1e-6, (cid, y0[cid], y1[cid])
tube = next(c for c in d1.data["components"] if c["id"] == "fin-001-l")
assert tube["geometry"]["params"]["center"][1] == -(int(span) / 2 - 70), tube["geometry"]["params"]["center"]
new = r.data["interference"]["new"]
r1 = rows(d1)
extra = sorted(set(r1) - set(r0))
print("interference.new:", len(new), "; T1 rows not in span-900 set:", len(extra), "; sets equal:", r1 == r0)
assert new == [] and extra == [], (new[:3], extra[:3])
if variant == "A":                              # example from the design: ger-001-ul y range [-390,-370] -> [-350,-330]
    assert [round(v, 6) for v in y0["ger-001-ul"]] == [-390.0, -370.0] and [round(v, 6) for v in y1["ger-001-ul"]] == [-350.0, -330.0]
print("OK")
```
