"""One-off: move finished units from plan section 7 to plans/<plan>/archive.md. Not committed."""
import sys
from pathlib import Path
from hangar.progress import UNIT_RE, parse_units

root = Path(sys.argv[1])
plan = root / "plans" / "2026-11-20-standardization.md"
lines = plan.read_text(encoding="utf-8").split("\n")
units = parse_units(lines)
finished = {u for u, d in units.items() if d["total"] and d["done"] == d["total"]}
in7 = False
blocks = []  # (start, end, uid)
cur = None
for i, line in enumerate(lines):
    if line.startswith("## "):
        if cur:
            blocks.append((cur[0], i, cur[1])); cur = None
        in7 = line.startswith("## 7.")
        continue
    if not in7:
        continue
    m = UNIT_RE.match(line)
    if m:
        if cur:
            blocks.append((cur[0], i, cur[1]))
        cur = (i, m.group(1))
if cur:
    blocks.append((cur[0], len(lines), cur[1]))
move = [b for b in blocks if b[2] in finished]
moved_text = []
drop = set()
for s, e, uid in move:
    moved_text.append("\n".join(lines[s:e]).rstrip("\n"))
    drop.update(range(s, e))
kept = [l for i, l in enumerate(lines) if i not in drop]
# pointer after the Conventions paragraph: insert right before the first remaining unit heading
for i, l in enumerate(kept):
    if UNIT_RE.match(l):

        break
plan.write_text("\n".join(kept), encoding="utf-8", newline="\n")

Do not edit.\n\n## 7. Archived units\n\n" + "\n\n".join(moved_text) + "\n", encoding="utf-8", newline="\n")
print(len(move), "moved:", ", ".join(b[2] for b in move)); print("kept units:", [b[2] for b in blocks if b[2] not in finished])
