```python
"""T6 mechanism gate (U3.6e): 1-degree sampling with range ends and 0, declared linkage pairs exempt, honest wording.

Collision volumes are faked (tests/fixtures/mech-good.yaml plus three boxes); only the sampling and the pair
logic are under test, not the mesh boolean."""
from __future__ import annotations

import copy
import datetime
from pathlib import Path

import pytest
import yaml

import hangar.checks
import hangar.geometry
from hangar.design import FAIL, PASS, Design
from hangar.gates import run_gates

FIX = Path(__file__).parent / "fixtures"
ROW = "T6.mechanism.flap_l"


def _box(cid: str, x: float) -> dict:
    return {"id": cid, "type": "other", "geometry": {"box": {"size": [4, 4, 4]}}, "position": [x, 20, 60], "mass_g": 1}


def make_design(*, rod_link: bool = True, input_range=(0, 20), joint_range=(0, 40)) -> Design:
    """mech-good (driver servo_arm -> arm; follower flap_hinge -> flap) plus rod (child of arm), horn (child of flap),
    blocker (child of wing)."""
    data = copy.deepcopy(yaml.safe_load((FIX / "mech-good.yaml").read_text(encoding="utf-8")))
    data["components"] += [_box("rod", 30), _box("horn", 260), _box("blocker", 150)]
    link = {"kinematic_link": True} if rod_link else {}
    data["attachments"] += [
        {"id": "arm-rod", "parent": "arm", "child": "rod", "method": "hinge", "location": [30, 20, 60],
         "rated_load_n": 5, **link},
        {"id": "flap-horn", "parent": "flap", "child": "horn", "method": "glued", "location": [260, 20, 60],
         "rated_load_n": 5},
        {"id": "wing-blocker", "parent": "wing", "child": "blocker", "method": "glued", "location": [150, 20, 60],
         "rated_load_n": 5},
    ]
    for a in data["attachments"]:
        if a["id"] in ("flap_hinge", "servo_arm"):
            a["articulation"]["range_deg"] = list(joint_range)
    data["mechanisms"][0]["input_range_deg"] = list(input_range)
    return Design(data, path=FIX / "mech-good.yaml")


def run_t6(monkeypatch, design: Design, volume_of):
    """Run tier 6 with fake overlap volumes. volume_of(frozenset({a, b}), driver_deg) -> mm3.
    Returns (row, driver angles posed, in order). Rest-pose overlaps are always 0."""
    state = {"owner": {}, "rest": set(), "driver": None, "posed": []}
    real_posed = hangar.geometry.posed_component_parts

    def posed(d, angles=None):
        out = real_posed(d, angles)
        for cid, parts in out.items():
            state["owner"][id(parts)] = cid
            if not angles:
                state["rest"].add(id(parts))
        if angles:
            state["driver"] = angles["servo_arm"]
            state["posed"].append(angles["servo_arm"])
        return out

    def overlap(a, b):
        if id(a) in state["rest"] or id(b) in state["rest"]:
            return 0.0
        return volume_of(frozenset((state["owner"][id(a)], state["owner"][id(b)])), state["driver"])

    monkeypatch.setattr(hangar.geometry, "posed_component_parts", posed)
    monkeypatch.setattr(hangar.checks, "_overlap_volume", overlap)
    card = run_gates(design, only_tiers={6}, today=datetime.date(2026, 11, 20))
    row = next(r for t in card.tiers for r in t.results if r.check_id == ROW)
    return row, state["posed"]


def test_one_degree_collision_is_found(monkeypatch):
    # collides only at input 3 deg (1.1 mm3 > the 1.0 limit): off a 5-degree grid
    row, _ = run_t6(monkeypatch, make_design(),
                    lambda pair, drv: 1.1 if pair == {"flap", "blocker"} and drv == 3.0 else 0.0)
    assert row.status == FAIL
    assert "flap and blocker collide at input 3 deg" in row.message


def test_linked_pair_is_not_counted_and_is_named(monkeypatch):
    row, _ = run_t6(monkeypatch, make_design(), lambda pair, drv: 5.0 if pair == {"rod", "horn"} else 0.0)
    assert row.status == PASS
    assert row.data["linked_pairs"] == [["horn", "rod"]]
    assert "linked pair" in row.message


@pytest.mark.parametrize("other", ["blocker", "flap"])
def test_unlinked_pairs_still_count(monkeypatch, other):
    row, _ = run_t6(monkeypatch, make_design(), lambda pair, drv: 5.0 if pair == {"rod", other} else 0.0)
    assert row.status == FAIL
    assert f"rod and {other}" in row.message or f"{other} and rod" in row.message


def test_pair_without_kinematic_link_still_counts(monkeypatch):
    row, _ = run_t6(monkeypatch, make_design(rod_link=False), lambda pair, drv: 5.0 if pair == {"rod", "horn"} else 0.0)
    assert row.status == FAIL
    assert "rod and horn" in row.message
    assert row.data["linked_pairs"] == []


def test_range_ends_zero_and_every_degree_are_sampled(monkeypatch):
    row, posed = run_t6(monkeypatch, make_design(input_range=(-3.5, 7.5), joint_range=(-10, 40)), lambda pair, drv: 0.0)
    assert row.status == PASS
    assert sorted(set(posed)) == [-3.5, -3, -2, -1, 0, 1, 2, 3, 4, 5, 6, 7, 7.5]
```

## Decisions (tactician proposals, for the conductor to confirm)
- (a) Step: the existing threshold key `mechanism_contact_step_deg` (source: internal convention) goes from 5.0 to 1.0; no new key, no literal in code. Both ends and 0 are always sampled (step 2). Cost: up to 5 times more boolean checks per mechanism; the concept-a probe (31 inputs per mechanism) ran in about a minute on the dev machine.
- (b) Exemption: exactly the pairs of (child of a `kinematic_link: true` attachment whose parent is on the driver side) x (parts moving with a driven joint, minus that joint's own child and minus anything on the driver side). On concept-a this is the single pair rod x horn per side, and the 1-degree probe found no other collision. The names are in `data["linked_pairs"]` and in the PASS message. `T1.interference` already skips `kinematic_link` child x moved pairs (`core/hangar/checks.py:458-469`) and reports them as one SKIP; the T6 exemption is narrower and reported per row. `standards/program.yaml:30` says "no exemptions (gates T6, T7)": this exemption is structural (a declared link), not a waiver; the conductor decides whether the program rule text needs a clause.
- (c) Tracking: the claim is dropped; the follower angle is stated as declared. Reason for not adding a geometric estimate: the schema has no field naming the two ends of the linkage (arm hole, horn hole). A scratch estimate from attachment locations using `E-MEC-03` (rod on the arm) and `E-MEC-PIN-R4` (pin on the horn) gives -1.28 against the mesh-measured -1.33 (3.8 percent apart, inside the 5 percent tolerance), but choosing `E-MEC-PIN-R4` automatically is not possible: the follower-side attachment nearest to the rod hole is `E-MEC-05` (the horn glue, 49 mm away against 51 mm), which is the wrong point. A rule that guesses the wrong point is worse than an honest statement. The measured lever ratio therefore stays a design-review item (the design-engineer puts the lever measurements in `--why`); a follow-up unit may add a linkage-ends field to the schema and then check the ratio. What the gate still proves: no jam, and no unlinked collision at 1 degree.
