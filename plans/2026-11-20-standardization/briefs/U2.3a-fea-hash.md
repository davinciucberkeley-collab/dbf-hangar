```python
"""Usage: rehash.py check|apply <BASE sha>. BASE = the commit the branch started from (.logs/base.txt)."""
import hashlib, json, pathlib, re, subprocess, sys
from hangar import fea
from hangar.design import REPO_ROOT, load_design

mode, base = sys.argv[1], sys.argv[2]
design = load_design(REPO_ROOT / "designs" / "concept-a" / "plane.yaml")
old_std = subprocess.run(["git", "-C", str(REPO_ROOT), "show", f"{base}:standards/design-standards.yaml"],
                         capture_output=True, check=True).stdout

def old_hash(pid):   # the pre-D13 formula, with the standards bytes as they were at BASE
    h = hashlib.sha256()
    h.update((design.base_dir / "plane.yaml").read_bytes())
    for f in sorted((design.base_dir / "meshes").glob("*")):
        if f.stem == pid:
            h.update(f.read_bytes())
    h.update(old_std)
    h.update(fea.RUNNER.read_bytes())
    return h.hexdigest()[:16]

same = subprocess.run(["git", "-C", str(REPO_ROOT), "diff", "--exit-code", "--stat", base, "--",
                       "designs/concept-a/plane.yaml", "designs/concept-a/meshes", "core/hangar/opencae_runner.py",
                       "standards/design-standards.yaml"]).returncode == 0
print("plane.yaml, meshes, runner and standards unchanged since BASE:", same)
bad, rows = [], []
for f in sorted((design.base_dir / "fea").glob("*.json")):
    stored = json.loads(f.read_text(encoding="utf-8"))["hash"]
    old, new = old_hash(f.stem), fea.part_hash(design, f.stem)
    rows.append((f, stored, old, new))
    print(f"{f.stem:<12} stored={stored} old_formula={old} new_formula={new} {'FRESH' if stored == old else 'ALREADY STALE'}")
    if stored != old:
        bad.append(f.stem)
if not same or bad:
    sys.exit(f"STOP: not a pure re-hash (inputs changed: {not same}; already stale: {bad})")
if mode == "apply":
    for f, stored, old, new in rows:
        text = f.read_bytes().decode("utf-8")
        out, n = re.subn(r'("hash": ")' + stored + '(")', r"\g<1>" + new + r"\g<2>", text, count=1)
        assert n == 1, f
        f.write_bytes(out.encode("utf-8"))
    print(f"rewrote {len(rows)} files")
```

```python
# Keys of OTHER components that the load cases read (geometry.py and sim.py). The part's own entry is hashed whole.
FEA_COMPONENT_KEYS = ("id", "type", "mass_g", "position", "rotation_deg", "cg_offset", "geometry", "cavities", "propulsion")


def part_inputs_fingerprint(design: Design, part_id: str) -> bytes:
    """Canonical bytes of the plane.yaml fields this part's result depends on: its own component entry, the load-relevant
    keys of every other component, all attachments (supports, load joints, welds), the root and the VTOL method."""
    data = design.data
    comps = [c if c["id"] == part_id else {k: c[k] for k in FEA_COMPONENT_KEYS if k in c} for c in design.components]
    picked = {"root": data.get("root"), "vtol_method": (data.get("morphology") or {}).get("vtol_method"),
              "components": comps, "attachments": data.get("attachments", [])}
    return json.dumps(picked, sort_keys=True, separators=(",", ":"), default=str).encode("utf-8")
```
