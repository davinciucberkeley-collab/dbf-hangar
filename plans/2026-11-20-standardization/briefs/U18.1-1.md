It imports only the standard library plus `from . import fea` and `from .design import Design`. Exact public interface (names and signatures are fixed; U18.1-2, -3 and -4 call them):

```python
CLI_ENV = ("OPENCAE_CLI_PYTHON", "OPENCAE_CLI_ROOT")
ANALYSES = ("mass", "propulsion", "structure", "cfd", "flight")

def find_cli(python: str | None = None, root: str | None = None) -> dict:
    """{python, root, found, how, head}. Order for each of python and root: argument, env var in CLI_ENV, keys cli_python / cli_root of fea.CONFIG_FILE (opencae.local.json), then
    auto. Auto root = the first folder among sorted(fea.REPO_ROOT.parent.glob("OpenCAEModel*")) that contains opencae/cli.py. Auto python = fea.find_opencae()["python"] (same environment search as FEA).
    found = python and root exist and (root/"opencae"/"cli.py").is_file(). head = first line of `git -C root rev-parse HEAD` or the version = "..." line of root/pyproject.toml, or "" ."""

def run_cli(cli: dict, args: list[str], timeout: int = 3600) -> tuple[int, dict | None, str]:
    """Run [cli["python"], "-m", "opencae", *args] with cwd=cli["root"], env PYTHONPATH=root + os.pathsep + existing, PYTHONIOENCODING=utf-8, PYTHONDONTWRITEBYTECODE=1, capture_output, text.
    Returns (returncode, parsed JSON object from stdout or None if stdout is not one JSON object, stderr tail of the last 3000 chars)."""

def doctor(cli: dict) -> dict:
    """run_cli(cli, ["doctor", "--json"], timeout=300). Returns the parsed object, or {"error": "<one line>"} when returncode != 0 or no JSON."""

def solver_status(doc: dict, environ: dict | None = None, home: Path | None = None) -> dict:
    """{"ccx": (bool, str), "cfd": (bool, str)}; the str is a reason when False, else what was found.
    ccx: doc["ccx"]["available"]. If doc has "error": both False with that text.
    cfd: True if doc["openfoam"]["available"]; else True if SU2_CFD found. SU2 search (the doctor does not report SU2): environ["SU2_RUN"], environ["OPENCAE_SU2"] (a file or a folder), shutil.which("SU2_CFD"), then any `**/bin/SU2_CFD.exe` (or `SU2_CFD` off Windows) under home/".opencae"/"solvers" and under Path("C:/SU2"). environ defaults to os.environ, home to Path.home().
    cfd False reason text: "no CFD solver: OpenFOAM not found and SU2_CFD not found; run: opencae setup su2 --yes (downloads about 40 MB from github.com/su2code/SU2, a human step)"."""

def cli_available() -> tuple[bool, str]:
    """(True, "OpenCAE CLI root=<root> python=<python> head=<head>") or (False, "OpenCAE CLI not found; run: plane opencae setup --cli-python <python> --cli-root <OpenCAEModel clone>")."""

def result_path(design: Design) -> Path:
    """design.base_dir / "opencae" / "results.json"."""

def project_hash(design: Design, project_dir: Path, analyses: tuple[str, ...], head: str) -> str:
    """First 16 hex of sha256 over, in this order: json.dumps(sorted(analyses)).encode(); (project_dir/"drone.json").read_bytes(); (project_dir/"airframe.stl").read_bytes(); head.encode(); STANDARDS_FINGERPRINT bytes."""

STANDARDS_KEYS = {"structure": ("ultimate_factor",), "operations": ("battery_usable_fraction", "mission_hover_min", "mission_cruise_min", "electrical_rating_min_ratio", "energy_reserve_fraction"),
                  "assumptions": ("hover_thrust_to_weight_min",)}  # see step 3: file of each key

def standards_fingerprint() -> bytes:
    """Canonical json (sort_keys, separators (",", ":")) bytes of the `value` of each key listed in STANDARDS_KEYS, read the way fea.standards_fingerprint reads design-standards keys and the way hangar/gates/discipline.py `_entry(OPS, key)` reads operations.yaml keys."""

def run_project(project_dir: Path, analyses: tuple[str, ...], cli: dict, log=print) -> dict:
    """Run `opencae run <project_dir> --only <a,b> --force --json` through run_cli. Returns {"returncode": int, "result": <parsed object or None>, "stderr_tail": str}. Does not write designs/. Raises ValueError for an analysis name not in ANALYSES."""

def design_fingerprint(design: Design, analyses: tuple[str, ...]) -> str:
    """Cheap staleness key a validate run can recompute without exporting or running anything: first 16 hex of sha256 over json.dumps(sorted(analyses)); canonical json (sort_keys, separators (",", ":"), default=str) of design.data; the bytes of design.base_dir/"opencae-inputs.yaml" if it exists; the bytes of every mesh file and generator file the design references, collected exactly as `fea._part_hash` collects them (core/hangar/fea.py lines 224-238: the `geometry.mesh.file` of each component, and `geom.registry.component_files(c)` for generator components), in sorted order, inside `with overlay.forced_off():` as fea.part_hash does; then standards_fingerprint()."""

def store_result(design: Design, hash_: str, design_hash: str, analyses: tuple[str, ...], head: str, result: dict) -> Path:
    """Write result_path(design) as json (indent 1, sort_keys False, trailing newline) with keys design (design.id), hash, design_hash, analyses (list), opencae_head, result. Creates the folder. Returns the path."""

def load_result(design: Design) -> dict | None:
    """The stored record, or None if the file is missing or not valid json."""
```
